    stored_layer(A, slot(B)).
predicate(linked_index/2)
linked_index(A, B) :-
    (   ground([B])
    ->  hash_consing:intern(slot(B), '__hash_consed_slot/1'(C)),
        stored_layer(A, link('__hash_consed_slot/1'(C), _))
    ;   (   ground(slot(B))
        ->  hash_consing:intern(slot(B), '__hash_consed_slot/1'(C))
        ;   true
        ),
        stored_layer(A, link('__hash_consed_slot/1'(C), _)),
        hash_consing:intern(slot(B), '__hash_consed_slot/1'(C))
    ).
